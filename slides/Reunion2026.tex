\begin{frame}{Blockchain?}

  A global database:
  \begin{itemize}
  \item transactions modify the state of the database
  \item transactions can be started by anybody, under payment
  \end{itemize}
  \bigskip
  The main properties of a blockchain:
  \begin{enumerate}
  \item immutability
  \item transparency
  \item censure-resistance
  \end{enumerate}

\end{frame}

\begin{frame}{Applications}
  Cryptocurrencies are just one of the applications of blockchain technology:
  \begin{itemize}
  \item decentralized money (Bitcoin, Ethereum, Cardano, Solana\ldots)
  \item decentralized smart contracts
  \item decentralized identities
  \item decentralized elections
  \item decentralized supply chains
  \item \ldots
  \end{itemize}
\end{frame}

\begin{frame}{A work of formalization}

  A formalization of proof of space requires to clarify:
  \begin{description}
  \item[Point~1:] what is the nonce data structure
  \item[Point~2:] how a challenge is derived from a block
  \item[Point~3:] how $\delay(\nonce,\challenge)$ is defined
  \end{description}

  \pause

  \medskip

  Once a formalization is available, it can be used to \alert{prove} that
  the algorithm is immune to some typical attacks and as the basis for
  a clean implementation.

  \bigskip

  \begin{greenbox}{}
    Fausto Spoto, \emph{A Formalization of Signum Consensus},
    Financial Cryptography Workshop on Trusted Smart Contracts,
    Miyakojima, Japan, 2025, volume 15753 of LNCS, pages 180--199.
  \end{greenbox}

\end{frame}

\begin{frame}{Point~1: The nonce data structure}
  \vspace*{-2ex}
  \[
  \begin{array}{c}
    \mathit{prolog}
    \left\{\begin{array}{l}
    \pi_{\mathit{node}}\in\mathit{byte}^*\text{ (public key of the node)}\\
    \pi_{\mathit{miner}}\in\mathit{byte}^*\text{ (public key of the miner)}\\
    \mathit{chainid}\text{ (name of the blockchain)}
    \end{array}\right.\\
    p\in\mathbb{N}\text{ (\emph{progressive})}
  \end{array}
  \]
  \[\begin{array}{l}
  \pause
  \seed=\mathit{prolog}\bowtie p\quad\text{($\bowtie$ is byte array concatenation)}\\
  \pause
  {\color{blue} h_0}=\hashfirstk(\seed)\\
  \pause
  {\color{blue} h_1}=\hashfirstk({\color{blue} h_0}\bowtie\seed)\\
  \pause
  {\color{blue} h_2}=\hashfirstk({\color{blue} h_1}\bowtie{\color{blue}  h_0}\bowtie \seed)\\
  \pause
  \cdots\\
  {\color{blue} h_{2n-1}}=\hashfirstk({\color{blue} h_{2n-2}}\bowtie{\color{blue} h_{2n-3}}\bowtie\cdots\bowtie{\color{blue} h_1}\bowtie{\color{blue} h_0}\bowtie \seed)\\
  \pause
  {\color{red}\finalhash}=\hash({\color{blue} h_{2n-1}}\bowtie{\color{blue} h_{2n-2}}\bowtie\cdots\bowtie{\color{blue} h_1}\bowtie{\color{blue} h_0}\bowtie \seed)\\
  \pause
  \text{reassign }{\color{blue} h_i}={\color{blue} h_i}\oplus{\color{red}\finalhash}\text{ for every }0\le i<2n\quad\text{($\oplus$ is byte array xor)}\\
  \end{array}\]

  \pause

  \[
  \nonce(p,\pi)=\langle p,\underbrace{{\color{blue}h_0},{\color{blue}h_{2n-1}}}_{\scoop_0},\underbrace{{\color{blue}h_2},{\color{blue}h_{2n-3}}}_{\scoop_1},\ldots,\underbrace{{\color{blue}h_{2n-4}},{\color{blue}h_3}}_{\scoop_{n-2}},\underbrace{{\color{blue}h_{2n-2}},{\color{blue}h_1}}_{\scoop_{n-1}}\rangle
  \]

  \begin{overprint}
    \onslide<10>
    \centering hash is the expensive shabal256
    \onslide<11>
    \centering $\kappa$ allows one to modulate the cost of the computation
    \onslide<12>
    \centering $\scoop_i={\color{blue}h_{2i}},{\color{blue}h_{2n-2i-1}}$ depends from at least a ${\color{blue}h_j}$ with $j\ge n$
  \end{overprint}

\end{frame}

\begin{frame}[fragile]{Point~2: Derivation of a challenge from a block}

  \begin{align*}
    {\color{blue}\challenge(\genesis)}&=\langle 0,\sigma_\genesis\rangle,\quad\sigma_\genesis\text{ is an arbitrary byte array}\\
    \mbox{}\\
    \visible<2->{
    {\color{blue}\challenge(\block)}&=\langle\hash(\sigma\bowtie{\color{red}\height})\ \mathit{mod}\ n,\sigma\rangle\\
    &\text{where $\sigma=\hash({\color{blue}\challenge(\previous)}.\sigma\bowtie{\color{red}\mathit{prolog}})$}}
  \end{align*}

  \visible<2->{
  \begin{center}
    \begin{tikzpicture}

    \end{tikzpicture}
  \end{center}
  }

  \visible<3->{
  \begin{center}
    \textbf{Proposition.} Challenges depends on the trunks only (simple proof by induction): proof of space has no block-grinding attacks.
  \end{center}
  }

\begin{frame}{Point~3: The delay of a nonce \wrt a challenge}

  \begin{center}
  Let $\challenge={\color{blue}\langle i,\sigma\rangle}$ and
  $\nonce=\langle p,\underbrace{h_0,h_{2n-1}}_{\scoop_0},\underbrace{h_2,h_{2n-3}}_{\scoop_1},\ldots,\underbrace{h_{2n-4},h_3}_{\scoop_{n-2}},\underbrace{h_{2n-2},h_1}_{\scoop_{n-1}}\rangle$:
  \end{center}
  \bigskip
  \[
    \delay(\nonce,\challenge)=\hash(\scoop_{\color{blue}i}\bowtie{\color{blue}\sigma})
  \]

\end{frame}

\begin{frame}[fragile]{Proof of space mining}
 
\begin{algorithm}[H] 
\caption{Proof of Space}
\begin{algorithmic}[1]
\Require{$block$, $plot$ file created for $\mathit{prolog}=\pi_{\mathit{node}},\pi_{\mathit{miner}},\mathit{chainid}$}
\Ensure{$next$ (a valid child of $block$)}
\Statex
\State {$best\_nonce$ $\gets$ first nonce of $plot$}
\State {$c$ $\gets$ $\challenge(block)$}
\For{$nonce \in plot$}
\If{$\delay(nonce, c)$ $<$ $\delay(best\_nonce, c)$}
\State {$best\_nonce$ $\gets$ $nonce$}
\EndIf
\EndFor
\State \Return {valid $next$ of $block$, created for $\mathit{prolog}$ and using $best\_nonce$}
\end{algorithmic}
\end{algorithm} 

\end{frame}

\begin{frame}\frametitle{Proof of Work vs.\ Proof of Space}

  \begin{greenbox}{Mining for Proof of Work}
    \begin{itemize}
    \item fully decentralized, high computational cost, minimal disk usage
    \item efficient code already exists (Bitcoin)
    \item remote mining is hard (large bandwidth)
    \item it needs significant investment, energy, good networking
    \item extremely unlikely in a poor country
    \end{itemize}
  \end{greenbox}

  \pause
  \bigskip

  \begin{greenbox}{Mining for Proof of Space}
    \begin{itemize}
    \item fully decentralized, minimal computational cost, high disk usage
    \item only vertical undocumented code exists
    \item remote mining is easy (minimal bandwidth)
    \item it needs no investment, almost no energy, minimal bandwidth
    \item possible also in poor countries
    \end{itemize}
  \end{greenbox}
\end{frame}

\begin{frame}\frametitle{Open questions}

  \begin{itemize}
  \item ASICs can recompute a plot file on-the-fly at each block challenge
    \begin{itemize}
    \item it is proof of work instead of proof of space
    \item not worthwhile as long as the crypto is cheap
    \item but what about the future?
    \end{itemize}
    \pause
  \item proof of space answers challenges with the \emph{best} available nonce
    \begin{itemize}
    \item can a suboptimal choice be used instead, to direct the system to
      subsequent challenges for which one has very good nonces available?
    \item it is proof of work again
    \item I already proved this impossible under some conditions, more general
      results are needed
    \end{itemize}
  \end{itemize}

\end{frame}
